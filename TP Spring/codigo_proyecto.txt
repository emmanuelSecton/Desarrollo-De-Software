export class FacturaService {
    private readonly apiBase = "http://localhost:8080/api";
    private readonly baseUrl = `${this.apiBase}/facturas`;

    async obtenerFacturas(filtros: FacturaFiltrosDTO = {}): Promise<PageResponseDTO<FacturaReporteDTO>> {
        const params = new URLSearchParams();

        if (filtros.numeroFactura !== undefined && !Number.isNaN(filtros.numeroFactura)) {
            params.append("numeroFactura", String(filtros.numeroFactura));
        }
        if (filtros.fechaDesde) params.append("fechaDesde", filtros.fechaDesde);
        if (filtros.fechaHasta) params.append("fechaHasta", filtros.fechaHasta);
        if (filtros.cliente && filtros.cliente.trim() !== "") {
            params.append("cliente", filtros.cliente.trim());
        }
        if (filtros.informadaArca === "true" || filtros.informadaArca === "false") {
            params.append("informadaArca", filtros.informadaArca);
        }
        params.append("page", String(filtros.page ?? 0));
        params.append("size", String(filtros.size ?? 5));

        const response = await fetch(`${this.baseUrl}?${params.toString()}`);
        if (!response.ok) {
            throw new Error(`Error al obtener facturas (HTTP ${response.status})`);
        }
        return await response.json();
    }

    async crearFactura(factura: FacturaRequestDTO): Promise<FacturaReporteDTO> {
        const response = await fetch(this.baseUrl, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(factura)
        });
        if (!response.ok) {
            const errorText = await response.text();
            throw new Error(errorText || `Error al crear factura (HTTP ${response.status})`);
        }
        return await response.json();
    }

    async actualizarEstado(id: number, estado: string, observaciones?: string): Promise<FacturaReporteDTO> {
        const body: FacturaRequestDTO = { estado };
        if (observaciones !== undefined) body.observaciones = observaciones;

        const response = await fetch(`${this.baseUrl}/${id}`, {
            method: "PUT",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body)
        });
        if (!response.ok) {
            const errorText = await response.text();
            throw new Error(`Error al actualizar factura (HTTP ${response.status}): ${errorText}`);
        }
        return await response.json();
    }

    async eliminarFactura(id: number): Promise<void> {
        const response = await fetch(`${this.baseUrl}/${id}`, { method: "DELETE" });
        if (!response.ok) {
            const errorText = await response.text();
            throw new Error(`Error al eliminar factura (HTTP ${response.status}): ${errorText}`);
        }
    }

    async informarArca(id: number): Promise<FacturaReporteDTO> {
        const response = await fetch(`${this.baseUrl}/${id}/informar-arca`, { method: "POST" });
        if (!response.ok) {
            const errorText = await response.text();
            throw new Error(`Error al informar a ARCA (HTTP ${response.status}): ${errorText}`);
        }
        return await response.json();
    }

    obtenerUrlPdf(id: number): string {
        return `${this.baseUrl}/${id}/pdf`;
    }

        const response = await fetch(`${this.apiBase}/clientes${query}`);

function fecha(iso?: string | null): string {
    if (!iso) return "-";
    const [a, m, d] = iso.substring(0, 10).split("-");
    return d && m && a ? `${d}/${m}/${a}` : iso;
}

function esc(valor: unknown): string {
    return String(valor ?? "").replace(/[&<>"']/g, (c) =>
        ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" })[c] as string
    );
}

function confirmar(texto: string, etiquetaOk = "Confirmar", peligro = false): Promise<boolean> {
    const btnSi = $<HTMLButtonElement>("btnConfirmarSi");
    const btnNo = $<HTMLButtonElement>("btnConfirmarNo");
    $<HTMLElement>("textoConfirmar").textContent = texto;
    btnSi.textContent = etiquetaOk;
    btnSi.className = peligro ? "danger-solid" : "";
    abrirModal(modalConfirmar, btnNo);

    return new Promise((resolve) => {
        const terminar = (resultado: boolean) => {
            cerrarModal(modalConfirmar);
            btnSi.removeEventListener("click", si);
            btnNo.removeEventListener("click", no);
            modalConfirmar.removeEventListener("cancelar", no);
            resolve(resultado);
        };
        const si = () => terminar(true);
        const no = () => terminar(false);
        btnSi.addEventListener("click", si);
        btnNo.addEventListener("click", no);
        modalConfirmar.addEventListener("cancelar", no);
    });
}

function subtotalItem(item: FacturaDetalleRequestDTO): number | null {
    if (item.precioUnitario === undefined) return null;
    return item.cantidad * item.precioUnitario * (1 - item.porcentajeBonificacion / 100);
}

function formatearCuit(valor: string): string {
    const d = valor.replace(/\D/g, "").substring(0, 11);

}

    public synchronized String[] obtenerTokenYSign() throws Exception {

            return new String[]{cachedToken, cachedSign};

        X509Certificate cert = cargarCertificado(certPath);

        ZonedDateTime now = ZonedDateTime.now();

        ZonedDateTime exp = now.plusMinutes(10);
        String expTime = exp.format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);

        String traXml = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
                "<loginTicketRequest version=\"1.0\">" +
                "<header>" +
                "<uniqueId>" + (System.currentTimeMillis() / 1000) + "</uniqueId>" +
                "<generationTime>" + genTime + "</generationTime>" +
                "<expirationTime>" + expTime + "</expirationTime>" +
                "</header>" +
                "<service>wsfe</service>" +
                "</loginTicketRequest>";

        CMSSignedDataGenerator gen = new CMSSignedDataGenerator();
        ContentSigner sha256Signer = new JcaContentSignerBuilder("SHA256withRSA")
                .setProvider("BC").build(privateKey);

        gen.addSignerInfoGenerator(new JcaSignerInfoGeneratorBuilder(
                new JcaDigestCalculatorProviderBuilder().setProvider("BC").build())
                .build(sha256Signer, cert));
        gen.addCertificates(new JcaCertStore(Collections.singletonList(cert)));

        CMSSignedData signedData = gen.generate(
                new CMSProcessableByteArray(traXml.getBytes(StandardCharsets.UTF_8)), true);
        String cmsBase64 = Base64.getEncoder().encodeToString(signedData.getEncoded());

        String soapEnvelope = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
                "<soapenv:Envelope xmlns:soapenv=\"http://schemas.xmlsoap.org/soap/envelope/\" " +
                "xmlns:wsaa=\"http://wsaa.view.sua.dvadac.desein.afip.gov\">" +
                "<soapenv:Header/>" +
                "<soapenv:Body><wsaa:loginCms><wsaa:in0>" + cmsBase64 + "</wsaa:in0></wsaa:loginCms></soapenv:Body>" +
                "</soapenv:Envelope>";

        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(wsaaUrl))
                .header("Content-Type", "text/xml; charset=utf-8")
                .header("SOAPAction", "")
                .POST(HttpRequest.BodyPublishers.ofString(soapEnvelope, StandardCharsets.UTF_8))
                .build();

        HttpResponse<String> response = httpClient.send(request, HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8));
        String xmlLimpio = response.body().replace("&lt;", "<").replace("&gt;", ">");

        if (response.statusCode() != 200 || !xmlLimpio.contains("<token>")) {
            throw new RuntimeException("Error en WSAA (HTTP " + response.statusCode() + "): " + xmlLimpio);
        }

        cachedToken = extraerTag(xmlLimpio, "token");
        cachedSign = extraerTag(xmlLimpio, "sign");
        cachedExpiration = exp;

        return new String[]{cachedToken, cachedSign};
    }

    private PrivateKey cargarClavePrivada(String path) throws Exception {
        try (PEMParser pemParser = new PEMParser(new FileReader(path))) {
            Object object = pemParser.readObject();
            JcaPEMKeyConverter converter = new JcaPEMKeyConverter().setProvider("BC");
            if (object instanceof PEMKeyPair keyPair) {
                return converter.getKeyPair(keyPair).getPrivate();
            } else if (object instanceof PrivateKeyInfo keyInfo) {
                return converter.getPrivateKey(keyInfo);
            }
            throw new IllegalArgumentException("Formato de clave privada no soportado: " + object);
        }
    }

    private X509Certificate cargarCertificado(String path) throws Exception {
        try (FileInputStream fis = new FileInputStream(path)) {
            return (X509Certificate) CertificateFactory.getInstance("X.509").generateCertificate(fis);
        }
    }

    private String extraerTag(String xml, String tag) {
        String open = "<" + tag + ">";
        String close = "</" + tag + ">";
        int start = xml.indexOf(open);
        int end = xml.indexOf(close);
        if (start == -1 || end == -1) return null;
        return xml.substring(start + open.length(), end).trim();
    }

        try {
            if (Desktop.isDesktopSupported() && Desktop.getDesktop().isSupported(Desktop.Action.BROWSE)) {
                Desktop.getDesktop().browse(new URI(url));
            } else if (System.getProperty("os.name").toLowerCase().contains("win")) {
                new ProcessBuilder("rundll32", "url.dll,FileProtocolHandler", url).start();
            } else {
                new ProcessBuilder("xdg-open", url).start();
            }
        } catch (Exception e) {
            System.err.println("[Frontend] No se pudo abrir el navegador: " + e.getMessage());
        }
    }

                Paths.get("frontend"),
                Paths.get("TP Spring", "frontend")

        String codigo = obligatorio(dto.getCodigo(), "El cÃ³digo").toUpperCase();
        String denominacion = obligatorio(dto.getDenominacion(), "La denominaciÃ³n");

        if (dto.getListaPrecioId() == null) {

        if (dto.getPrecioVenta() == null || dto.getPrecioVenta() <= 0) {

        }

        Long existentes = em.createQuery(
                        "SELECT COUNT(a) FROM Articulo a WHERE UPPER(a.codigo) = :codigo", Long.class)
                .setParameter("codigo", codigo)
                .getSingleResult();
        if (existentes > 0) {
            throw new ResponseStatusException(HttpStatus.CONFLICT, "Ya existe un artÃ­culo con el cÃ³digo " + codigo + ".");
        }

        ListaPrecio lista = buscar(ListaPrecio.class, dto.getListaPrecioId(), "Lista de precios");
        Rubro rubro = dto.getRubroId() != null ? buscar(Rubro.class, dto.getRubroId(), "Rubro") : null;
        Marca marca = dto.getMarcaId() != null ? buscar(Marca.class, dto.getMarcaId(), "Marca") : null;

        auditar(precio, usuario);

    }

    @Transactional
    public ClienteDTO crear(ClienteRequestDTO dto) {

        String cuit = normalizarCuit(obligatorio(dto.getCuitCuil(), "El CUIT/CUIL"));

        String email = opcional(dto.getEmail());

        if (email != null && !EMAIL.matcher(email).matches()) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "El email no tiene un formato vÃ¡lido.");
        }

        Long existentes = em.createQuery(
                        "SELECT COUNT(c) FROM Cliente c WHERE REPLACE(c.cuitCuil, '-', '') = :digitos", Long.class)
                .setParameter("digitos", cuit.replace("-", ""))
                .getSingleResult();
        if (existentes > 0) {

        }

        Usuario usuario = obtenerUsuarioPorDefecto();
        Date ahora = new Date();

    public PageResponseDTO<FacturaReporteDTO> buscarFacturasPaginadas(
            Long numeroFactura,
            Date fechaDesde,
            Date fechaHasta,
            String cliente,
            Boolean informadaArca,
            int page,
            int size
    ) {
        StringBuilder whereClause = new StringBuilder("WHERE 1=1 ");
        Map<String, Object> params = new HashMap<>();

        if (numeroFactura != null) {
            whereClause.append("AND f.numero = :numeroFactura ");
            params.put("numeroFactura", numeroFactura);
        }
        if (fechaDesde != null) {
            whereClause.append("AND f.fechaEmision >= :fechaDesde ");
            params.put("fechaDesde", fechaDesde);
        }
        if (fechaHasta != null) {
            whereClause.append("AND f.fechaEmision <= :fechaHasta ");
            params.put("fechaHasta", fechaHasta);
        }
        if (cliente != null && !cliente.trim().isEmpty()) {
            whereClause.append("AND LOWER(COALESCE(c.denominacion, 'Consumidor Final')) LIKE LOWER(:cliente) ");
            params.put("cliente", "%" + cliente.trim() + "%");
        }
        if (informadaArca != null) {
            if (informadaArca) {
                whereClause.append("AND f.cae IS NOT NULL AND f.cae <> '' ");
            } else {
                whereClause.append("AND (f.cae IS NULL OR f.cae = '') ");
            }
        }

        // 1. Contar total de facturas que cumplen los filtros
        String countJpql = "SELECT COUNT(DISTINCT f.id) FROM FacturaVenta f LEFT JOIN f.cliente c " + whereClause;
        TypedQuery<Long> countQuery = em.createQuery(countJpql, Long.class);
        params.forEach(countQuery::setParameter);
        long totalElements = countQuery.getSingleResult();

        // 2. Traer solo la pÃ¡gina solicitada
        String dataJpql =
            "SELECT new ar.edu.utn.tp2.dto.FacturaReporteDTO(" +
            "f.id, f.numero, f.fechaEmision, COALESCE(c.denominacion, 'Consumidor Final'), " +
            "ci.denominacion, pv.descripcion, f.importeTotal, COUNT(d), f.estado, " +
            "f.cae, f.caeFechaVencimiento, f.resultadoAfip" +
            ") " +
            "FROM FacturaVenta f " +
            "LEFT JOIN f.cliente c " +
            "LEFT JOIN f.condicionIva ci " +
            "LEFT JOIN f.puntoVenta pv " +
            "LEFT JOIN f.detalles d " +
            whereClause +
            "GROUP BY f.id, f.numero, f.fechaEmision, c.denominacion, ci.denominacion, pv.descripcion, f.importeTotal, f.estado, f.cae, f.caeFechaVencimiento, f.resultadoAfip " +
            "ORDER BY f.id DESC";

        int safePage = Math.max(page, 0);
        int safeSize = size > 0 ? size : 5;

        TypedQuery<FacturaReporteDTO> dataQuery = em.createQuery(dataJpql, FacturaReporteDTO.class);
        params.forEach(dataQuery::setParameter);
        dataQuery.setFirstResult(safePage * safeSize);
        dataQuery.setMaxResults(safeSize);

        List<FacturaReporteDTO> content = dataQuery.getResultList();
        int totalPages = (int) Math.ceil((double) totalElements / safeSize);

        return new PageResponseDTO<>(content, safePage, safeSize, totalElements, totalPages);
    }

    @Transactional
    public FacturaReporteDTO crear(FacturaRequestDTO dto) {
        if (dto.getDetalles() == null || dto.getDetalles().isEmpty()) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "La factura debe contener al menos un detalle.");
        }

        PuntoVenta puntoVenta = buscarEntidadObligatoria(PuntoVenta.class, dto.getPuntoVentaId(), "Punto de venta");

        // El nÃºmero de factura es Ãºnico dentro de cada punto de venta (como en ARCA)
        if (dto.getNumero() != null) {
            Long existentes = em.createQuery(
                            "SELECT COUNT(f) FROM FacturaVenta f WHERE f.numero = :numero AND f.puntoVenta = :puntoVenta",
                            Long.class)
                    .setParameter("numero", dto.getNumero())
                    .setParameter("puntoVenta", puntoVenta)
                    .getSingleResult();

                        "Ya existe una factura con el nÃºmero " + dto.getNumero()
                                + " en el punto de venta " + puntoVenta.getNumero() + ".");
            }
        }
        CondicionIva condicionIva = buscarEntidadObligatoria(CondicionIva.class, dto.getCondicionIvaId(), "CondiciÃ³n de IVA");

        Cliente cliente = null;
        if (dto.getClienteId() != null) {
            cliente = buscarEntidadObligatoria(Cliente.class, dto.getClienteId(), "Cliente");
        }

        Usuario usuario = obtenerUsuarioPorDefecto();

        FacturaVenta factura = new FacturaVenta();
        factura.setNumero(dto.getNumero());
        factura.setFechaEmision(dto.getFechaEmision() != null ? dto.getFechaEmision() : new Date());

        factura.setCondicionIva(condicionIva);
        factura.setTipoMoneda(tipoMoneda);
        factura.setPuntoVenta(puntoVenta);
        factura.setEstado(dto.getEstado() != null && !dto.getEstado().isBlank() ? dto.getEstado() : "EMITIDA");
        factura.setObservaciones(dto.getObservaciones());
        factura.setDetalles(new ArrayList<>());

        Date ahora = new Date();
        factura.setFechaAlta(ahora);
        factura.setFechaModificacion(ahora);

        factura.setUsuarioModificac(usuario);

        factura.setImporteCobrado(0.0);

    @Transactional
    public FacturaReporteDTO informarArca(Long id) {
        FacturaVenta factura = em.find(FacturaVenta.class, id);
        if (factura == null) {
            throw new ResponseStatusException(HttpStatus.NOT_FOUND, "Factura no encontrada con ID: " + id);
        }
        if (estaInformadaArca(factura)) {
            throw new ResponseStatusException(HttpStatus.CONFLICT, "La factura ya fue informada a ARCA (CAE: " + factura.getCae() + ").");
        }
        if ("ANULADA".equalsIgnoreCase(factura.getEstado())) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "No se puede informar a ARCA una factura ANULADA.");
        }

        try {
            String[] auth = wsaaClient.obtenerTokenYSign();
            String cuitCliente = factura.getCliente() != null ? factura.getCliente().getCuitCuil() : null;
            int ptoVta = factura.getPuntoVenta().getNumero();

            AfipCaeResponse resp = wsfeClient.solicitarCae(
                    auth[0], auth[1], ptoVta, factura.getFechaEmision(), factura.getImporteTotal(), cuitCliente
            );

            factura.setResultadoAfip(resp.getResultado());

            if ("A".equalsIgnoreCase(resp.getResultado()) && resp.getCae() != null && !resp.getCae().isBlank()) {
                factura.setCae(resp.getCae());
                if (resp.getCaeFechaVencimiento() != null && !resp.getCaeFechaVencimiento().isBlank()) {
                    Date vto = new SimpleDateFormat("yyyyMMdd").parse(resp.getCaeFechaVencimiento());
                    factura.setCaeFechaVencimiento(vto);
                }
                factura.setNumero(resp.getNumeroComprobante());
                factura.setMotivoRechazo(null);
                factura.setFechaModificacion(new Date());
                em.merge(factura);
                em.flush();
                return mapearAReporteDTO(factura);
            } else {
                String motivo = resp.getObservaciones() != null ? resp.getObservaciones() : "Comprobante rechazado por ARCA.";
                throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "ARCA rechazÃ³ la factura: " + motivo);
            }
        } catch (ResponseStatusException e) {
            throw e;
        } catch (Exception e) {
            throw new ResponseStatusException(HttpStatus.INTERNAL_SERVER_ERROR, "Error al comunicarse con ARCA: " + e.getMessage());
        }
    }

    @Transactional
    public FacturaReporteDTO actualizar(Long id, FacturaRequestDTO dto) {
        FacturaVenta factura = em.find(FacturaVenta.class, id);
        if (factura == null) {
            throw new ResponseStatusException(HttpStatus.NOT_FOUND, "Factura no encontrada con ID: " + id);
        }

        // Regla de negocio: si ya estÃ¡ informada a ARCA, bloquear modificaciones (solo permitir Anular)
        if (estaInformadaArca(factura)) {
            boolean intentaModificarDatos = dto.getNumero() != null
                    || dto.getFechaEmision() != null
                    || dto.getClienteId() != null
                    || dto.getCondicionIvaId() != null
                    || dto.getPuntoVentaId() != null
                    || dto.getTipoMonedaId() != null
                    || (dto.getDetalles() != null && !dto.getDetalles().isEmpty());

            boolean esAnulacion = "ANULADA".equalsIgnoreCase(dto.getEstado());

            if (intentaModificarDatos || !esAnulacion) {
                throw new ResponseStatusException(
                        HttpStatus.CONFLICT,
                        "La factura ya fue informada a ARCA (CAE: " + factura.getCae() + "). No se puede modificar, solo Anular."
                );
            }

            factura.setEstado("ANULADA");
            if (factura.getFechaAnulacion() == null) {
                factura.setFechaAnulacion(new Date());
            }
            if (dto.getObservaciones() != null) {
                factura.setObservaciones(dto.getObservaciones());
            }
            factura.setFechaModificacion(new Date());
            em.merge(factura);
            em.flush();
            return mapearAReporteDTO(factura);
        }

        // Si NO estÃ¡ informada a ARCA, permite modificar cabecera e Ã­tems normalmente
        if (dto.getNumero() != null) factura.setNumero(dto.getNumero());
        if (dto.getFechaEmision() != null) factura.setFechaEmision(dto.getFechaEmision());
        if (dto.getClienteId() != null) factura.setCliente(buscarEntidadObligatoria(Cliente.class, dto.getClienteId(), "Cliente"));
        if (dto.getCondicionIvaId() != null) factura.setCondicionIva(buscarEntidadObligatoria(CondicionIva.class, dto.getCondicionIvaId(), "CondiciÃ³n de IVA"));
        if (dto.getPuntoVentaId() != null) factura.setPuntoVenta(buscarEntidadObligatoria(PuntoVenta.class, dto.getPuntoVentaId(), "Punto de venta"));
        if (dto.getTipoMonedaId() != null) factura.setTipoMoneda(buscarEntidadObligatoria(TipoMoneda.class, dto.getTipoMonedaId(), "Tipo de moneda"));
        if (dto.getObservaciones() != null) factura.setObservaciones(dto.getObservaciones());

        if (dto.getEstado() != null && !dto.getEstado().isBlank()) {
            factura.setEstado(dto.getEstado());
            if ("ANULADA".equalsIgnoreCase(dto.getEstado()) && factura.getFechaAnulacion() == null) {
                factura.setFechaAnulacion(new Date());
            }
        }

        if (dto.getDetalles() != null && !dto.getDetalles().isEmpty()) {
            factura.getDetalles().clear();
            double nuevoTotal = cargarDetallesYCalcularTotal(factura, dto.getDetalles());
            factura.setImporteTotal(nuevoTotal);
            factura.setImporteSaldo(nuevoTotal - factura.getImporteCobrado());
        }

        factura.setFechaModificacion(new Date());
        em.merge(factura);
        em.flush();

        return mapearAReporteDTO(factura);
    }

    private double cargarDetallesYCalcularTotal(FacturaVenta factura, List<FacturaDetalleRequestDTO> detallesDTO) {
        double totalFactura = 0.0;

        for (FacturaDetalleRequestDTO itemDto : detallesDTO) {
            ListaPrecioArticulo lpa = buscarListaPrecioPorArticuloId(itemDto.getArticuloId());

            double precioUnitario = (itemDto.getPrecioUnitario() != null && itemDto.getPrecioUnitario() > 0)
                    ? itemDto.getPrecioUnitario()
                    : lpa.getPrecioVenta();

            double bruto = itemDto.getCantidad() * precioUnitario;
            double descuento = bruto * (itemDto.getPorcentajeBonificacion() / 100.0);
            double importeNeto = bruto - descuento;
            double importeIva = importeNeto * ALICUOTA_IVA;
            double importeSubtotal = importeNeto + importeIva;

            FacturaVentaDetalle detalle = new FacturaVentaDetalle();
            detalle.setFactura(factura);
            detalle.setListaPrecioArticulo(lpa);
            detalle.setDescripcion(
                    itemDto.getDescripcion() != null && !itemDto.getDescripcion().isBlank()
                            ? itemDto.getDescripcion()
                            : lpa.getArticulo().getDenominacion()
            );
            detalle.setCantidad(itemDto.getCantidad());
            detalle.setPrecioUnitario(precioUnitario);
            detalle.setPorcentajeBonificacion(itemDto.getPorcentajeBonificacion());
            detalle.setImporteNeto(importeNeto);
            detalle.setImporteIva(importeIva);
            detalle.setImporteSubtotal(importeSubtotal);

            factura.getDetalles().add(detalle);
            totalFactura += importeSubtotal;
        }

        return totalFactura;
    }

        }

    }

                    "No se encontrÃ³ un precio registrado (ListaPrecioArticulo) para el artÃ­culo ID: " + articuloId

        return resultados.get(0);
    }

import jakarta.persistence.PersistenceContext;
import jakarta.transaction.Transactional;
import org.springframework.boot.CommandLineRunner;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;

@Component
@Order(2)
public class ReporteInicialLoader implements CommandLineRunner {

    @PersistenceContext
    private EntityManager entityManager;

    @Override
    @Transactional
    public void run(String... args) {
        Long cantidadFacturas = entityManager
                .createQuery("SELECT COUNT(f) FROM FacturaVenta f", Long.class)
                .getSingleResult();

        if (cantidadFacturas > 0) {
            ConsultasJPQL.generarReporteDTO(entityManager);
        }
    }
}
